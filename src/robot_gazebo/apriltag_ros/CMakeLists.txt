cmake_minimum_required(VERSION 3.8)
project(apriltag_ros)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(apriltag REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/AprilTagDetection.msg" "msg/AprilTagDetectionArray.msg"
  DEPENDENCIES std_msgs geometry_msgs)
add_executable(apriltag_node src/apriltag_node.cpp)
target_compile_features(apriltag_node PUBLIC cxx_std_17)
target_compile_options(apriltag_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(apriltag_node rclcpp sensor_msgs geometry_msgs cv_bridge tf2 tf2_ros)
target_link_libraries(apriltag_node apriltag::apriltag ${OpenCV_LIBRARIES})
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(apriltag_node "${typesupport}")
install(TARGETS apriltag_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/check_apriltag.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
